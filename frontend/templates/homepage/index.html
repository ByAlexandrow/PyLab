def greet(name):
    """Greet a user with a welcome message"""
    return f"Hello, {name}! Welcome to PyLab."

def calculate_stats(numbers):
    """Calculate basic statistics"""
    if not numbers:
        return None
    
    stats = {
        'sum': sum(numbers),
        'average': sum(numbers) / len(numbers),
        'min': min(numbers),
        'max': max(numbers)
    }
    return stats

# Your code here
